# TT030 bootstrap floppy (FR-6): mounts BlueSCSI/internal SCSI partitions and loads STiNG.
# <dest on A:>            <source, repo-relative>
# AUTO/ runs in THIS order: SCSIDRV (SCSI cookie) -> STiNG (SCSILINK needs SCSIDRV) -> ICDBOOT LAST.
# ICDBOOT must be last: after installing it runs the hard disk's C:\AUTO\*.PRG itself and the
# rest of A:\AUTO never runs (measured in Hatari 2026-09-11 with --trace gemdos: with ICDBOOT
# before STING.PRG, STING.PRG was never Pexec'd; with it after, all four STX modules load).
# 00JAR032.PRG removed 2026-09-11: the 64-byte "cookie jar expander" ends in a bare RTS, so TOS
# kills it with 11 bombs (Pterm(-1) in the Hatari trace). TOS 3.06 has its own cookie jar.
AUTO/STING.INF             floppy/STING.INF              # one line: A:\STING\
AUTO/SCSIDRV.PRG           staging/DRIVERS/SCSI/CBHD/cbhd502/CBHD502/SCSIDRV.PRG   # CBHD 5.02 SCSIDRV cookie; SCSILINK.STX (WiFi) needs it
AUTO/STING.PRG             staging/AUTO/STING.PRG        # STiNG 1.26 kernel
AUTO/HELPSPC.PRG           staging/AUTO_EXTRA/HELPSPC.PRG   # keys right of L and P type space (space bar dead); must precede ICDBOOT, which ends A:\AUTO
AUTO/ICDBOOT.PRG           staging/AUTO/ICDBOOT.PRG      # ICD AdSCSI Pro 6.5.5 (freeware, C3)

# STiNG loads every *.STX in the folder named by STING.INF.
STING/DEFAULT.CFG          floppy/DEFAULT.CFG            # distribution template, ACTIVATE = TRUE
STING/ROUTE.TAB            floppy/ROUTE.TAB              # Modem 1 SLIP routes to fractal 192.168.10.1
STING/TCP.STX              staging/STING/TCP.STX
STING/UDP.STX              staging/STING/UDP.STX
STING/RESOLVE.STX          staging/STING/RESOLVE.STX
STING/SCSILINK.STX         staging/AUTO_EXTRA/SCSILINK.STX   # Anodyne DaynaPORT driver (BlueSCSI ID 3)
STING/SERIAL.STX           staging/STING/SERIAL.STX      # SLIP/PPP on Modem 1 -> fractal /dev/atari-tt (null modem)
STING/PING.PRG             staging/STING/TOOLS/PING.PRG

# Port IP config is done on the TT in the Control Panel (STNGPORT.CPX).
XCONTROL.ACC               staging/XCONTROL.ACC
CPX/STING.CPX              staging/CPX/STING.CPX
CPX/STNGPORT.CPX           staging/CARD_SAVED_2026-09-12/CPX/STNGPORT.CPX   # as saved on the TT: Ethernet0 192.168.0.30/24
CPX/SERIAL.CPX             staging/CPX/SERIAL.CPZ        # Modem 1 baud rate (CPZ = disabled name; enabled here)
STING.PRT                  staging/CARD_SAVED_2026-09-12/STING.PRT   # STiNG port settings saved by STNGPORT on the TT
